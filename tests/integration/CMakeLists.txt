# Integration tests grouped by mathematical domain (see subdirectories).
# Target names stay short so MSVC object paths fit CMAKE_OBJECT_PATH_MAX.
file(GLOB_RECURSE _MS_INTEGRATION_SOURCES RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
    "*.cpp")
list(SORT _MS_INTEGRATION_SOURCES)
foreach(_src ${_MS_INTEGRATION_SOURCES})
    get_filename_component(_stem ${_src} NAME_WE)
    get_filename_component(_dir ${_src} DIRECTORY)
    if(_dir STREQUAL "")
        set(_name ${_stem})
    else()
        string(REPLACE "/" "_" _dir ${_dir})
        string(REGEX REPLACE "^test_" "" _short ${_stem})
        set(_name "int_${_dir}_${_short}")
        string(LENGTH "${_name}" _nlen)
        if(_nlen GREATER 72)
            string(MD5 _hash "${_src}")
            string(SUBSTRING "${_hash}" 0 10 _hid)
            set(_name "int_${_dir}_${_hid}")
        endif()
    endif()
    add_ms_test(${_name} ${_src})
    if(_stem STREQUAL "test_mathscriptc_pipeline")
        target_compile_definitions(${_name} PRIVATE "MATHSCRIPTC_PATH=\"$<TARGET_FILE:mathscriptc>\"")
        add_dependencies(${_name} mathscriptc)
    endif()
endforeach()
